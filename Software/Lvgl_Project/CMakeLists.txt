cmake_minimum_required(VERSION 3.12.4)
if(POLICY CMP0123)
    cmake_policy(SET CMP0123 NEW)
    set(CMAKE_POLICY_DEFAULT_CMP0123 NEW)
endif()
if(POLICY CMP0177)
    cmake_policy(SET CMP0177 NEW)
    set(CMAKE_POLICY_DEFAULT_CMP0177 NEW)
endif()
project(BikeComputerUI LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UI_PLATFORM "PC" CACHE STRING "UI platform: PC or STM32")
set_property(CACHE UI_PLATFORM PROPERTY STRINGS PC STM32)
if(UI_PLATFORM STREQUAL "PC")
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/pc.cmake)
elseif(UI_PLATFORM STREQUAL "STM32")
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/stm32.cmake)
else()
    message(FATAL_ERROR "Unknown UI_PLATFORM: ${UI_PLATFORM}")
endif()
